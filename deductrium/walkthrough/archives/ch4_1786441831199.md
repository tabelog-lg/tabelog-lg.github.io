## 外延公理与分离公理
### 外延公理
- 外延公理
- aExt:(∀x:(∀y:((∀z:¬(((z∈x)→(z∈y))→¬((z∈y)→(z∈x))))→(x=y))))
- 门是它的简化版
- .aext:(∀x:(∀y:((∀z:((z∈x)↔(z∈y)))→(x=y))))
- 简化流程
```
d aExt
d vvvd<>1 x y z z@x z@y
d vv<a5 1
d vv.t 2 0
```
- 也是十分简单
- 解锁了分离公理aSep:(∀#nf(\$x,\$y,\$z):¬(∀#nf(\$y,\$x,\$z):¬(∀#nf(\$z,\$x,\$y):¬(((#nf(\$z,\$x,\$y)∈#nf(\$y,\$x,\$z))→¬((#nf(\$z,\$x,\$y)∈#nf(\$x,\$y,\$z))→¬#nf(\$0,\$y)))→¬(¬((#nf(\$z,\$x,\$y)∈#nf(\$x,\$y,\$z))→¬#nf(\$0,\$y))→(#nf(\$z,\$x,\$y)∈#nf(\$y,\$x,\$z)))))))
- 这么长肯定是要简化的 
- 简化流程
```
d aSep $x $y $z $0
d v:dE,.<>s #nf($x,$y,$z) #nf($y,$x,$z) (V#nf($z,$x,$y):~(((#nf($z,$x,$y)@#nf($y,$x,$z))>~((#nf($z,$x,$y)@#nf($x,$y,$z))>~#nf($0,$y)))>~(~((#nf($z,$x,$y)@#nf($x,$y,$z))>~#nf($0,$y))>(#nf($z,$x,$y)@#nf($y,$x,$z)))))
d .<>rV 1
d d vvv:d&,.<>s #nf($x,$y,$z) #nf($y,$x,$z) #nf($z,$x,$y) (#nf($z,$x,$y)@#nf($y,$x,$z))>~((#nf($z,$x,$y)@#nf($x,$y,$z))>~#nf($0,$y)) ~((#nf($z,$x,$y)@#nf($x,$y,$z))>~#nf($0,$y))>(#nf($z,$x,$y)@#nf($y,$x,$z))
d vv.<>rV 3
d v.<>rE 4
d .<>rV 5
d vvv.<>i #nf($x,$y,$z) #nf($y,$x,$z) #nf($z,$x,$y) #nf($z,$x,$y)@#nf($y,$x,$z)
d vvv:d&,.<>s #nf($x,$y,$z) #nf($y,$x,$z) #nf($z,$x,$y) #nf($z,$x,$y)@#nf($x,$y,$z) #nf($0,$y)
d vvv<>r> 7 8
d vvv:d&,.<>s #nf($x,$y,$z) #nf($y,$x,$z) #nf($z,$x,$y) #nf($z,$x,$y)@#nf($x,$y,$z) #nf($0,$y)
d vvv.<>i #nf($x,$y,$z) #nf($y,$x,$z) #nf($z,$x,$y) #nf($z,$x,$y)@#nf($y,$x,$z)
d vvv.<>r> 10 11
d vvv.<>r& 9 12
d vv.<>rV 13
d v.<>rE 14
d .<>rV 15
d vvv:.<>&,.<>s #nf($x,$y,$z) #nf($y,$x,$z) #nf($z,$x,$y) #nf($z,$x,$y)@#nf($y,$x,$z) (#nf($z,$x,$y)@#nf($x,$y,$z))&#nf($0,$y)
d vv.<>rV 17
d v.<>rE 18
d .<>rV 19
d .<>t 2 6
d .<>t 21 16
d .<>t 22 20

```
